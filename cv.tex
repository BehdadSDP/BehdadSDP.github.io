	\section{Publications}
	
	\begin{enumerate}[label={[\arabic*]}]
		\item \textbf{A Deep Learning-Enhanced Liquid Crystal Sensor for the Rapid Discrimination of Snake Venoms: A Case Study on Cobra and Viper}, \textit{Sensors and Actuators A: Physical, Elsevier} -- Published -- \href{https://doi.org/10.1016/j.sna.2026.117779}{DoI} \hfill (2026) \\
		Sahar S. Mousavi Sani, Mohammad Mohammadimasoudi*, \textbf{Behdad Sadeghian Pour}, Mahdi Behdani, Delavar Shahbazzadeh \\
		\textit{Developed a low-cost liquid crystal sensor on a fabric textile grid to rapidly distinguish cobra and viper venoms. The system utilizes a lightweight MobileNetV3 deep learning model to automatically identify venom-induced visual pattern changes. It achieves a 95\% classification accuracy in just 4 minutes with a detection limit of 45 $\mu$g/mL.}
		
		\item \textbf{AL-YOLO: Accurate and Lightweight Vehicle and Pedestrian Detector in Foggy Weather}, \textit{14th International Conference on Computer and Knowledge Engineering, IEEE} -- Published -- \href{https://doi.org/10.1109/ICCKE65377.2024.10874685}{DoI} \hfill (2024) \\
		\textbf{Behdad Sadeghian Pour}; Hamidreza Mohammadi Jozani; Shahriar Baradaran Shokouhi \\
		\textit{Introduced AL-YOLO, a lightweight object detector optimized for identifying vehicles and pedestrians in low-visibility foggy weather. By combining Mobile ViTv2 and an Inverted Residual Block, it enhances feature extraction efficiency for resource-constrained devices. The model achieves a 33\% parameter reduction and a 2\% accuracy improvement over YOLOv5s.}
		
		\item \textbf{CNN and ViT as Dual Teachers for Knowledge Distillation in Human Action Recognition}, \textit{10th International Conference on Signal Processing \& Intelligent Systems Shahrood University of Technology, IEEE} -- Published -- \href{https://doi.org/10.1109/ICSPIS65223.2024.10931096}{DoI} \hfill (2024) \\
		Hamidreza Mohammadi Jozani; \textbf{Behdad Sadeghian Pour}; Shahriar B. Shokouhi \\
		\textit{Proposed a deep learning framework that improves human action recognition in static images using knowledge distillation. A small student model learns from dual teachers—a CNN for local details and a Vision Transformer for global context—integrated via an adaptive loss-aware module. This method achieved a top performance of 95.17\% mean Average Precision on the Stanford40 dataset.}
		
		\item \textbf{Knowledge Distillation with Reused Teacher Classifier Framework for Action Recognition in Still Image}, \textit{Preprint} \hfill (2024)
	\end{enumerate}
